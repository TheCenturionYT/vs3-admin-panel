This file must use `locals.pb` (the member-scoped PocketBase client) for ALL queries. Never use an admin client. The PocketBase collection rule `@request.auth.faction = faction` on the nodes collection enforces faction privacy at the DB level when the member token is used.

**Full load function:**

```typescript
import { error } from '@sveltejs/kit';
import type { PageServerLoad } from './$types';
import { calcUpkeep } from '$lib/upkeep';

export const load: PageServerLoad = async ({ locals, parent }) => {
  const { user } = await parent();
  const factionId = user.factionId;

  if (!factionId) {
    error(500, 'Faction not found');
  }

  // 4 parallel queries — all via locals.pb (member token, never admin)
  const [faction, rawNodes, wars, alliances] = await Promise.all([
    locals.pb.collection('factions').getOne(factionId),
    locals.pb.collection('nodes').getFullList({
      filter: 'faction = {:factionId} && owner_type != "neutral"',
      filterValues: { factionId },
      sort: 'name'
    }),
    locals.pb.collection('wars').getFullList({
      filter: 'status = "active"',
      expand: 'faction_a,faction_b',
      sort: '-created'
    }),
    locals.pb.collection('alliances').getFullList({
      filter: 'status = "active"',
      expand: 'factions',
      sort: '-created'
    })
  ]);

  if (!faction) {
    error(500, 'Faction not found');
  }

  // Count active wars involving this faction (for war upkeep modifier)
  const factionWarCount = wars.filter(
    (w) => w.faction_a === factionId || w.faction_b === factionId
  ).length;

  const nodeCount = rawNodes.length;

  // Compute effective upkeep and cycle payment status for each node
  const nodes = rawNodes.map((node) => {
    const effectiveUpkeep = calcUpkeep(
      node.base_upkeep,
      nodeCount,
      factionWarCount,
      faction.type as 'PvP' | 'PvE',
      false // never neutral — filtered above
    );

    // paidSP: sum of current cycle submissions
    // Field name from Phase 3 schema: node.cycle_paid_sp (number, updated by scheduler/submissions)
    const paidSP: number = node.cycle_paid_sp ?? 0;
    const requiredSP: number = effectiveUpkeep;
    const paymentPct = requiredSP > 0 ? paidSP / requiredSP : 0;

    let upkeepStatus: 'Paid' | 'Partial' | 'Underfunded' | 'Unpaid';
    if (paymentPct >= 1) upkeepStatus = 'Paid';
    else if (paymentPct >= 0.5) upkeepStatus = 'Partial';
    else if (paymentPct > 0) upkeepStatus = 'Underfunded';
    else upkeepStatus = 'Unpaid';

    // Military node tier labels
    const MILITARY_TIER_LABELS: Record<number, string> = {
      1: 'Watchtower', 2: 'Outpost', 3: 'Fort', 4: 'Bastion'
    };
    const isMilitary = node.type === 'Military';
    const tierLabel = isMilitary ? MILITARY_TIER_LABELS[node.tier] ?? '' : '';

    return {
      id: node.id,
      name: node.name,
      type: node.type as string,
      tier: node.tier as number,
      isMilitary,
      tierLabel,
      instability: node.instability as number,
      upkeepStatus,
      paidSP,
      requiredSP,
      paymentPct
    };
  });

  // Shape wars for portal display
  const warsForPortal = wars.map((w) => ({
    id: w.id,
    factionAName: (w.expand?.faction_a?.name ?? w.faction_a) as string,
    factionBName: (w.expand?.faction_b?.name ?? w.faction_b) as string,
    casusBelli: (w.casus_belli ?? '') as string,
    startDate: w.start_date as string  // ISO string, formatted in UI
  }));

  // Shape alliances for portal display
  const alliancesForPortal = alliances.map((a) => {
    const parties: string[] = Array.isArray(a.expand?.factions)
      ? (a.expand.factions as Array<{ name: string }>).map((f) => f.name)
      : [];
    return {
      id: a.id,
      type: a.type as string,
      parties,
      startDate: a.start_date as string
    };
  });

  return {
    faction: {
      id: faction.id,
      name: faction.name as string,
      type: faction.type as 'PvP' | 'PvE'
    },
    nodes,
    wars: warsForPortal,
    alliances: alliancesForPortal
  };
};
```

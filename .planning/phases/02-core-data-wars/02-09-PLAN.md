use:enhance={() => {
  return async ({ result }) => {
    if (result.type === 'success' && result.data?.exportJson) {
      const blob = new Blob([result.data.exportJson], { type: 'application/json' });
      const url = URL.createObjectURL(blob);
      const a = document.createElement('a');
      a.href = url;
      a.download = `vs3-backup-${new Date().toISOString().slice(0,10)}.json`;
      a.click();
      URL.revokeObjectURL(url);
    }
  };
}}

```javascript
routerAdd("GET", "/api/vs3/export", (e) => {
    const result = { exported_at: new Date().toISOString() };
    const collections = [
        'factions', 'faction_members', 'nodes', 'node_ownership_history',
        'wars', 'battles', 'sieges', 'diplomacy', 'server_log'
    ];
    for (const col of collections) {
        try {
            result[col] = $app.dao().findRecordsByFilter(col, '', '-created', 0, 0);
        } catch (err) {
            console.error('[export] Could not fetch ' + col + ':', err);
            result[col] = [];
        }
    }
    return e.json(200, result);
}, $apis.requireAuth());
```

`exportData` action:
```typescript
exportData: async ({ locals, fetch }) => {
  const pbUrl = process.env.POCKETBASE_URL ?? 'http://localhost:8090';
  const response = await fetch(`${pbUrl}/api/vs3/export`, {
    headers: { Authorization: locals.pb.authStore.token }
  });
  if (!response.ok) {
    return fail(500, { action: 'exportData', errors: { _global: ['Export failed. Check PocketBase connection.'] } });
  }
  const json = await response.json();
  return { success: true, action: 'exportData', exportJson: JSON.stringify(json, null, 2) };
}
```

```typescript
const [factions, nodes, activeWars] = await Promise.all([
  locals.pb.collection('factions').getFullList({ fields: 'id,is_system', sort: 'name' }),
  locals.pb.collection('nodes').getFullList({
    expand: 'owner',
    sort: '-instability',
    fields: 'id,name,type,tier,instability,owner,expand'
  }),
  locals.pb.collection('wars').getFullList({
    filter: 'status = "active"',
    expand: 'faction_a,faction_b',
```

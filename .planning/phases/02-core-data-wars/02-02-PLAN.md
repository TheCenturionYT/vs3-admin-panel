```typescript
export function overextensionMul(nodeCount: number): number

// PvE: always 0. PvP: 0wars→0, 1war→0.15, 2wars→0.30, 3+→0.50

export function calcUpkeep(
  baseUpkeep: number,
  nodeCount: number,
  warCount: number,
  factionType: 'PvP' | 'PvE',
  isNeutral: boolean
): number
// isNeutral || !baseUpkeep → return baseUpkeep as-is
// Otherwise: Math.ceil(baseUpkeep * overextensionMul(nodeCount) * (1 + warMul(warCount, factionType)))
```
